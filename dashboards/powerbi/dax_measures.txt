// DISPLAY FOLDER: 01 - Executive KPIs
// ========================================================================================

/// 1. Total distinct operational orders placed across network
[Total Orders] = 
DISTINCTCOUNT('FactOrders'[order_id])
// Format: #,##0
// Description: Unfiltered distinct count of all operational orders placed across the network.

/// 2. Count of successfully fulfilled and delivered orders
[Delivered Orders] = 
CALCULATE(
    [Total Orders],
    KEEPFILTERS('FactOrders'[order_status] = "Delivered")
)
// Format: #,##0
// Description: Total volume of orders with completed delivery status (27,727 baseline).

/// 3. Count of cancelled orders prior to delivery completion
[Cancelled Orders] = 
CALCULATE(
    [Total Orders],
    KEEPFILTERS('FactOrders'[order_status] = "Cancelled")
)
// Format: #,##0
// Description: Total volume of orders cancelled across network (252 baseline).

/// 4. Percentage of total orders cancelled
[Cancellation Rate %] = 
DIVIDE(
    [Cancelled Orders],
    [Total Orders],
    0
)
// Format: 0.00%
// Description: Proportion of network orders cancelled relative to total intake.

/// 5. On-Time Delivery Rate (OTD %) — Executive SLA Standard
[On-Time Delivery Rate %] = 
DIVIDE(
    [On-Time Deliveries],
    [Delivered Orders],
    0
)
// Format: 0.00%
// Description: Primary customer SLA metric. Denominator strictly uses Delivered Orders (excludes cancelled).

/// 6. Delivery Delay Rate %
[Delay Rate %] = 
DIVIDE(
    [Delayed Deliveries],
    [Delivered Orders],
    0
)
// Format: 0.00%
// Description: Proportion of delivered orders that breached scheduled delivery window (20.4% baseline).

/// 7. Average delivery transit duration in hours
[Avg Delivery Duration (Hours)] = 
CALCULATE(
    AVERAGE('FactOrders'[delivery_time_hours]),
    KEEPFILTERS('FactOrders'[order_status] = "Delivered")
)
// Format: 0.00
// Description: Average hours elapsed from order dispatch to final customer drop-off.

/// 8. Average Customer Satisfaction (CSAT) score (1.00 - 5.00 scale)
[Avg CSAT Score] = 
CALCULATE(
    AVERAGE('FactOrders'[csat_score]),
    KEEPFILTERS('FactOrders'[order_status] = "Delivered")
)
// Format: 0.00
// Description: Average customer rating recorded upon delivery confirmation.

/// 9. Order volume in prior calendar month (Time Intelligence)
[Total Orders (Prior Month)] = 
CALCULATE(
    [Total Orders],
    DATEADD('DimDate'[full_date], -1, MONTH)
)
// Format: #,##0
// Description: Evaluates total orders shifted back by one calendar month.

/// 10. Month-over-Month percentage order volume growth
[Order Volume MoM Growth %] = 
VAR _Current = [Total Orders]
VAR _Prior   = [Total Orders (Prior Month)]
RETURN
    DIVIDE(_Current - _Prior, _Prior, 0)
// Format: +0.00%;-0.00%;0.00%
// Description: Period-over-period order intake growth percentage.


// ========================================================================================
// DISPLAY FOLDER: 02 - Delivery & SLAs
// ========================================================================================

/// 11. Count of successfully delivered orders meeting SLA deadline
[On-Time Deliveries] = 
CALCULATE(
    [Total Orders],
    KEEPFILTERS('FactOrders'[order_status] = "Delivered"),
    KEEPFILTERS('FactOrders'[is_on_time] = TRUE())
)
// Format: #,##0
// Description: Orders successfully delivered on or before the committed delivery window.

/// 12. Count of successfully delivered orders breaching SLA deadline
[Delayed Deliveries] = 
CALCULATE(
    [Total Orders],
    KEEPFILTERS('FactOrders'[order_status] = "Delivered"),
    KEEPFILTERS('FactOrders'[is_delayed] = TRUE())
)
// Format: #,##0
// Description: Orders delivered past scheduled commitment window (5,653 baseline).

/// 13. Average transit duration for delayed orders
[Avg Delayed Duration (Hours)] = 
CALCULATE(
    AVERAGE('FactOrders'[delivery_time_hours]),
    KEEPFILTERS('FactOrders'[order_status] = "Delivered"),
    KEEPFILTERS('FactOrders'[is_delayed] = TRUE())
)
// Format: 0.00
// Description: Mean delivery duration for orders experiencing operational delays.

/// 14. Average transit duration for on-time orders
[Avg On-Time Duration (Hours)] = 
CALCULATE(
    AVERAGE('FactOrders'[delivery_time_hours]),
    KEEPFILTERS('FactOrders'[order_status] = "Delivered"),
    KEEPFILTERS('FactOrders'[is_on_time] = TRUE())
)
// Format: 0.00
// Description: Mean delivery duration for orders completed on schedule.

/// 15. Operational delivery duration variance between delayed and on-time shipments
[Delivery SLA Variance (Hours)] = 
[Avg Delayed Duration (Hours)] - [Avg On-Time Duration (Hours)]
// Format: +0.00;-0.00;0.00
// Description: Net time penalty incurred by delayed shipments.

/// 16. Total delay incidents categorized by root-cause reason
[Delay Incidents by Reason] = 
CALCULATE(
    COUNTROWS('FactOrders'),
    KEEPFILTERS(NOT ISBLANK('FactOrders'[delay_reason])),
    KEEPFILTERS('FactOrders'[order_status] = "Delivered")
)
// Format: #,##0
// Description: Count of order delay occurrences attributed to specific root-cause categories.

/// 17. Pareto cumulative percentage across ranked delay reasons
[Cumulative Delay Reason %] = 
VAR _CurrentCount = [Delay Incidents by Reason]
VAR _TotalIncidents = CALCULATE([Delay Incidents by Reason], ALLSELECTED('FactOrders'[delay_reason]))
VAR _TableRanked = 
    ADDCOLUMNS(
        ALLSELECTED('FactOrders'[delay_reason]),
        "@Incidents", [Delay Incidents by Reason]
    )
VAR _CumulativeSum = 
    SUMX(
        FILTER(_TableRanked, [@Incidents] >= _CurrentCount),
        [@Incidents]
    )
RETURN
    DIVIDE(_CumulativeSum, _TotalIncidents, 0)
// Format: 0.0%
// Description: Pareto 80/20 cumulative percentage line for delay root-cause analysis.

/// 18. Percentage share of total delay incidents for current category
[Delay Share %] = 
DIVIDE(
    [Delay Incidents by Reason],
    CALCULATE([Delay Incidents by Reason], ALL('FactOrders'[delay_reason])),
    0
)
// Format: 0.0%
// Description: Percentage share of delay incidents contributed by the selected category.

// Description: Unique commercial vehicles that have undergone servicing.

// DISPLAY FOLDER: 07 - Dynamic Visual Formatting & Heatmap Colors

/// 55. Dynamic font color highlighting hub with highest delay rate
[Delay Rate Font Color] = 
VAR CurrentDelay = [Delay Rate %]
VAR MaxDelay = 
    MAXX(
        ALLSELECTED('DimHub'),
        CALCULATE([Delay Rate %])
    )
RETURN
    IF(
        CurrentDelay = MaxDelay,
        "#FB7185", // High-risk highlight (Rose/Red)
        "#FFFFFF"  // Standard white
    )

/// 56. Dynamic font color highlighting hub with highest on-time delivery rate
[On-Time Rate Font Color] = 
VAR CurrentRate = [On-Time Delivery Rate %]
VAR MaxRate = 
    MAXX(
        ALLSELECTED('DimHub'),
        CALCULATE([On-Time Delivery Rate %])
    )
RETURN
    IF(
        CurrentRate = MaxRate,
        "#34D399", // Peak performance highlight (Emerald Green)
        "#FFFFFF"
    )

/// 57. Dynamic driver leaderboard color: Top on-time performer
[On-Time Delivery Rate Font Color] = 
VAR DriverRates = 
    ADDCOLUMNS(
        ALLSELECTED('DimDriver'),
        "@Rate", CALCULATE([On-Time Delivery Rate %])
    )
VAR MaxRate = MAXX(DriverRates, [@Rate])
VAR WinningDriver = 
    MINX(
        FILTER(DriverRates, [@Rate] = MaxRate),
        'DimDriver'[driver_name]
    )
VAR CurrentDriver = SELECTEDVALUE('DimDriver'[driver_name])
RETURN
    IF(
        NOT ISBLANK(CurrentDriver) && CurrentDriver = WinningDriver,
        "#34D399", // Emerald Green for top performer
        "#FFFFFF"
    )

/// 58. Dynamic driver leaderboard color: Volume leader
[Delivered Orders Font Color] = 
VAR DriverOrders = 
    ADDCOLUMNS(
        ALLSELECTED('DimDriver'),
        "@Orders", CALCULATE([Delivered Orders])
    )
VAR MaxOrders = MAXX(DriverOrders, [@Orders])
VAR WinningDriver = 
    MINX(
        FILTER(DriverOrders, [@Orders] = MaxOrders),
        'DimDriver'[driver_name]
    )
VAR CurrentDriver = SELECTEDVALUE('DimDriver'[driver_name])
RETURN
    IF(
        NOT ISBLANK(CurrentDriver) && CurrentDriver = WinningDriver,
        "#FBBF24", // Amber Gold for volume leader
        "#FFFFFF"
    )

/// 59. Dynamic driver leaderboard color: Top quality rating
[Performance Rating Font Color] = 
VAR CurrentRating = SELECTEDVALUE('DimDriver'[performance_rating])
VAR MaxRating = 
    MAXX(
        ALLSELECTED('DimDriver'),
        CALCULATE(MAX('DimDriver'[performance_rating]))
    )
RETURN
    IF(
        NOT ISBLANK(CurrentRating) && CurrentRating = MaxRating,
        "#38BDF8", // Cyan Blue for rating leader
        "#FFFFFF"
    )

/// 60. Dynamic alert color for incident feed delayed records
[Order ID Font Color] = 
VAR DelayedValue = [Delay Incidents by Reason]
RETURN
    IF(
        NOT ISBLANK(DelayedValue) && DelayedValue > 0,
        "#EF4444", // Red warning for delayed record
        "#FFFFFF"
    )

/// 61. Dynamic ML prediction label color
[ML Order ID Font Color] = 
VAR CurrentLabel = SELECTEDVALUE('FactMLPredictions'[predicted_label])
RETURN
    SWITCH(
        CurrentLabel,
        "DELAY_RISK", "#EF4444", // High-risk red
        "ON_TIME",    "#34D399", // Low-risk emerald
        "#FFFFFF"
    )
